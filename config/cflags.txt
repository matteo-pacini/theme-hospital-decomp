# First matching glob (relative to the repo root) decides how a source file
# is compiled: glob, compiler (optionally @version: 10.6 by default, or
# 10.0a), flags. -zq, -hd, -zm, -fr and -i=include are
# always added; the pseudo-flag -zm- drops -zm for a module built with all its
# functions in one segment (tools/split.py then keeps them in one section).
#
# Bullfrog libraries (optimised C++); flags still being established. Must
# precede the game rule, which would also match these paths.
# The network module netint is built like the other network code (no -4).
# netint: -oe=100 (94..103 all match) with the module's internal functions static, as the
# demo marks them: a static function called once is inlined whatever its size, so
# Start/CompleteMultiPlayerExchange and GenericIPXInit inline, while LbNetwork_Init(u32, ...)
# and HostData* stay out of line by size.
src/lib/bf1/opt/netint.cpp  wpp386  -s -oneatx -xs -oe=100 -zm-
src/lib/bf1/opt/netint3.cpp  wpp386  -s -oneatx -xs
# fade_sam: one segment (FadeSample's jump table follows fn_000ccd30 directly).
src/lib/bf1/opt/fade_sam.cpp  wpp386  -s -oneatx -4 -xs -zm-
src/lib/bf1/opt/*.cpp  wpp386  -s -oneatx -4 -xs
src/lib/bf1/*.cpp    wpp386  -s -d2 -otx
src/lib/bf2/dbg/*.cpp    wpp386  -s -oneatx -d1+
src/lib/bf2/snd/*.cpp    wpp386  -s -oneatx -4 -xs
# StopMusic stores its constants as immediates: no instruction scheduling (-or).
src/lib/bf2/stop_mus.cpp  wpp386  -s -oneatli -xs
# dosser: one segment; SerialHandler::Init's body follows the constructor while
# its entry follows CommHandler::Init, which falls through into it.
src/lib/bf2/dosser.cpp  wpp386  -s -oneatx -xs -zm-
# init_sou: built for the 486 like the other sound modules (lea for the scaled index).
src/lib/bf2/init_sou.cpp  wpp386  -s -oneatx -4 -xs
# serialsp: inlines the small Parser accessors into ProcessAvailableMsg but not SendAck.
# One segment: its code ends with the no-op filler that aligns the inline
# SerialSP destructor (a COMDAT) after it.
src/lib/bf2/serialsp.cpp  wpp386  -s -oneatx -xs -oe=40 -zm-
# modemsp: one segment, like serialsp (filler before the inline ModemSP destructor).
src/lib/bf2/modemsp.cpp  wpp386  -s -oneatx -xs -zm-
# dosipx: RecPkt stays out of line in UpdateServer only below -oe=90 (the
# network modules otherwise agree on plain -oe); ours is likely a few nodes smaller.
src/lib/bf2/dosipx.cpp  wpp386  -s -oneatx -xs -oe=80
src/lib/bf2/*.cpp    wpp386  -s -oneatx -xs

# Watcom C/C++ runtime (clib, C++ support): small model, segment registers
# saved (-r), -ox without a/t; the C++ support code for the 386 (-3) and flat
# model (-mf: inline string instructions assume es == ds).
# stat was built for the 386 and for size (enter/leave, movzx).
src/lib/clib/stat.c  wcc386  -s -oxs -ms -r -3
# qsort returns through med3's epilogue (built without -zm).
src/lib/clib/qsort.c wcc386  -s -oxs -ms -r -3 -zm-
# scnf: scan_string and scan_float end in a jump to scan_char's epilogue;
# FP conversions are calls (-fpc).
src/lib/clib/scnf.c  wcc386  -s -ox -ms -r -fpc -zm-
# asctime: ctime falls through into asctime.
src/lib/clib/asctime.c  wcc386  -s -ox -ms -r -zm-
# Built without -zm: flushall and fcloseall fall through into the next
# function, __set_EDOM into __set_errno, _dosretax into __set_errno_dos.

src/lib/clib/fclosall.c  wcc386  -s -ox -ms -r -zm-

src/lib/clib/dosret.c    wcc386  -s -ox -ms -r -zm-

# Bullfrog's Smacker player wrapper: optimised C++ for the Pentium (-5: no loop

# RAD's Smacker library: C, same optimisation (-oe inlines its helpers). Its

# Game code: C++, full debug info (which also disables optimisation). -ot
# must come first: -d2 turns optimisation off but keeps the "optimise for
# time" preference, which decides register allocation.
# alloc.cpp holds global objects with constructors: its static initialiser
# registers them for exception cleanup (-xs).
src/alloc.cpp        wpp386  -s -ot -d2 -xs
