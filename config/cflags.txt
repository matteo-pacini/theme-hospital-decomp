# First matching glob (relative to the repo root) decides how a source file
# is compiled: glob, compiler, flags. -zq, -hd, -zm, -fr and -i=include are
# always added; the pseudo-flag -zm- drops -zm for a module built with all its
# functions in one segment (tools/split.py then keeps them in one section).
#
# Bullfrog libraries (optimised C++); flags still being established. Must
# precede the game rule, which would also match these paths.
# The network modules (ServiceProvider, netint) are built like the other network code (no -4).
src/lib/bf1/opt/sp.cpp  wpp386  -s -oneatx -xs
# netint: -oe=400 inlines Start/CompleteMultiPlayerExchange, GenericIPXInit and the exchange
# functions into LbNetwork_Exchange as the original does; the original value is not established
# (it also inlines LbNetwork_Init(u32, ...), which the original does not).
src/lib/bf1/opt/netint.cpp  wpp386  -s -oneatx -xs -oe=400
src/lib/bf1/opt/netint3.cpp  wpp386  -s -oneatx -xs
src/lib/bf1/opt/*.cpp  wpp386  -s -oneatx -4 -xs
src/lib/bf1/*.cpp    wpp386  -s -d2 -otx
src/lib/bf2/dbg/*.cpp    wpp386  -s -oneatx -d1+
src/lib/bf2/snd/*.cpp    wpp386  -s -oneatx -4 -xs
# StopMusic stores its constants as immediates: no instruction scheduling (-or).
src/lib/bf2/stop_mus.cpp  wpp386  -s -oneatli -xs
# dosser: one segment; SerialHandler::Init's body follows the constructor while
# its entry follows CommHandler::Init, which falls through into it.
src/lib/bf2/dosser.cpp  wpp386  -s -oneatx -xs -zm-
src/lib/bf2/*.cpp    wpp386  -s -oneatx -xs
# Watcom C/C++ runtime (clib, C++ support): small model, segment registers
# saved (-r), -ox without a/t; the C++ support code for the 386 (-3) and flat
# model (-mf: inline string instructions assume es == ds).
# stat was built for the 386 and for size (enter/leave, movzx).
src/lib/clib/stat.c  wcc386  -s -oxs -ms -r -3
# qsort returns through med3's epilogue (built without -zm).
src/lib/clib/qsort.c wcc386  -s -oxs -ms -r -3 -zm-
# scnf: scan_string and scan_float end in a jump to scan_char's epilogue;
# FP conversions are calls (-fpc).
src/lib/clib/scnf.c  wcc386  -s -ox -ms -r -fpc -zm-
# asctime: ctime falls through into asctime.
src/lib/clib/asctime.c  wcc386  -s -ox -ms -r -zm-
# Built without -zm: flushall and fcloseall fall through into the next
# function, __set_EDOM into __set_errno, _dosretax into __set_errno_dos.
src/lib/clib/flushall.c  wcc386  -s -ox -ms -r -zm-
src/lib/clib/fclosall.c  wcc386  -s -ox -ms -r -zm-
src/lib/clib/seterrno.c  wcc386  -s -ox -ms -r -zm-
src/lib/clib/dosret.c    wcc386  -s -ox -ms -r -zm-
# itoa/ltoa read their digit table through cs: (based on _CODE), which only
# works when the functions share _TEXT with it.
src/lib/clib/ltoa.c      wcc386  -s -ox -ms -r -zm-
src/lib/clib/itoa.c      wcc386  -s -ox -ms -r -zm-
# prtf: far_other_strlen returns through far_strlen's epilogue.
src/lib/clib/prtf.c      wcc386  -s -ox -ms -r -zm-
# tzset tail-calls into its neighbours; __parse_tz is split around the
# parsing helpers.
src/lib/clib/tzset.c     wcc386  -s -ox -ms -r -zm-
# grownear: __FreeDPMIBlocks and __CreateNewNHeap return through
# RationalAlloc's epilogue.
src/lib/clib/grownear.c  wcc386  -s -ox -ms -r -zm-
# strtol jumps into strtoul.
src/lib/clib/strtol.c    wcc386  -s -ox -ms -r -zm-
# The FP support routines keep a frame pointer (-of); _Scale10V checks for
# the Pentium FDIV bug (-fpd).
src/lib/clib/scale.c   wcc386  -s -ox -ms -r -of -fpd
src/lib/clib/log87e.c  wcc386  -s -ox -ms -r -of
src/lib/clib/cvt.c     wcc386  -s -ox -ms -r -of
src/lib/clib/strtod.c  wcc386  -s -ox -ms -r -of
src/lib/clib/math2err.c wcc386  -s -ox -ms -r -of
# ftos: DoFFormat returns through DoEFormat's epilogue (built without -zm).
src/lib/clib/ftos.c    wcc386  -s -ox -ms -r -of -zm-
src/lib/clib/*.c     wcc386  -s -ox -ms -r
# stabdtor: __wcpp_2_destruct_all_ and __wcpp_2_destruct_after_all_ fall
# through into the function after them.
src/lib/clib/stabdtor.cpp  wpp386  -s -ox -mf -r -3 -zm-
# stabtrav: __wcpp_2_stab_trav_move_ ends in a jump to
# __wcpp_2_stab_trav_next_'s epilogue.
src/lib/clib/stabtrav.cpp  wpp386  -s -ox -mf -r -3 -zm-
src/lib/clib/*.cpp   wpp386  -s -ox -mf -r -3
# Bullfrog's Smacker player wrapper: optimised C++.
src/lib/smk/*.cpp    wpp386  -s -oneatx -4
# RAD's Smacker library: C, same optimisation (-oe inlines its helpers).
src/lib/smk/*.c      wcc386  -s -oneatx -oe=25 -5
# Game code: C++, full debug info (which also disables optimisation). -ot
# must come first: -d2 turns optimisation off but keeps the "optimise for
# time" preference, which decides register allocation.
src/*.cpp            wpp386  -s -ot -d2
# Miles Sound System (RAD Game Tools): C, optimised for the 486, public
# functions are __cdecl.

# falls through into rbpos.
src/lib/ail/ailcd.c  wcc386  -s -oxat -4 -zm-
src/lib/ail/*.c      wcc386  -s -oxat -4
