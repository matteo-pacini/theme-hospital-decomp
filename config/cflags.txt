# First matching glob (relative to the repo root) decides how a source file
# is compiled: glob, compiler, flags. -zq, -hd, -zm and -i=include are always
# added.
#
# Bullfrog libraries (optimised C++); flags still being established. Must
# precede the game rule, which would also match these paths.
src/lib/bf1/opt/*.cpp  wpp386  -s -oneatx -4 -xs
src/lib/bf1/*.cpp    wpp386  -s -d2 -otx
src/lib/bf2/dbg/*.cpp    wpp386  -s -oneatx -d1+
src/lib/bf2/*.cpp    wpp386  -s -oneatx -xs
# Watcom C/C++ runtime (clib, C++ support): small model, -ox without a/t.
# -nd=clib only lengthens the data segment names: objconv 2.54 writes
# section symbols pointing past a too-short COFF string table (objects with
# only short symbol names) and aborts with "Pointer out of range".
src/lib/clib/*.c     wcc386  -s -ox -ms -nd=clib
src/lib/clib/*.cpp   wpp386  -s -oxat
# Game code: C++, full debug info (which also disables optimisation). -ot
# must come first: -d2 turns optimisation off but keeps the "optimise for
# time" preference, which decides register allocation.
src/*.cpp            wpp386  -s -ot -d2
# Miles Sound System (RAD Game Tools): C, optimised for the 486, public
# functions are __cdecl.
src/lib/ail/*.c      wcc386  -s -oxat -4
# Modules that were hand-written assembly (*.asm in the demo's debug info),
# as GNU as source generated by tools/asm2src.py.
src/*.s              as
